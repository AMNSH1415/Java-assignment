Question 4:

Age Category Take the user's age and display the appropriate category: - 0–12: Child - 13–19: Teenager - 20–59: Adult - 60 and above: Senior Citizen

Solution

let age = Number(prompt("Enter your age:")); if (age >= 0 && age <= 12) { console.log("Child"); } else if (age >= 13 && age <= 19) { console.log("Teenager"); } else if (age >= 20 && age <= 59) { console.log("Adult"); } else if (age >= 60) { console.log("Senior Citizen");}

Question 5:

Electricity Bill Calculator Take the number of electricity units consumed and calculate the bill using these rates: - First 100 units: Rs. 10 per unit - Next 100 units: Rs. 15 per unit - Above 200 units: Rs. 20 per unit

Solution

let units = Number(prompt("Enter electricity units:")); let bill; if (units < 0) { console.log("Invalid Units"); } else if (units <= 100) { bill = units * 10; console.log("Electricity Bill: Rs.", bill); } else if (units <= 200) { bill = (100 * 10) + ((units - 100) * 15); console.log("Electricity Bill: Rs.", bill); } else { bill = (100 * 10) + (100 * 15) + ((units - 200) * 20); console.log("Electricity Bill: Rs.", bill); }

Question 6:

Traffic Light Take a traffic light color as input. - Red: `"Stop"` - Yellow: `"Ready"` - Green: `"Go"` - Any other color: `"Invalid Color"`

Solution

let color = prompt("Enter traffic light color:"); if (color.toLowerCase() === "red") { console.log("Stop"); } else if (color.toLowerCase() === "yellow") { console.log("Ready"); } else if (color.toLowerCase() === "green") { console.log("Go"); } else { console.log("Invalid Color"); }
